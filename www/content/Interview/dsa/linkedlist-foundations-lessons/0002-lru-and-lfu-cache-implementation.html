LRU & LFU Cache from Scratch

Architecting O(1) get, put, and eviction caches combining linked lists and hash tables.

Track goal: Master foundational data structures from 0 level. This is Lesson 2 of 2 in LinkedList Foundations. Index

1 · The One Idea

An LRU (Least Recently Used) Cache must support two operations in strictly O(1) time: get(key) and put(key, value). If the cache exceeds capacity, it must evict the least recently used key. A HashMap alone gives $O(1)$ lookups but has no ordering. A LinkedList maintains order but takes $O(N)$ to find an element. By combining both (HashMap stores pointers to Doubly Linked List nodes), we get the superpower of $O(1)$ lookups combined with $O(1)$ node removals and head re-insertions!

2 · Complete Java Implementation of LRU Cache

import java.util.HashMap;
import java.util.Map;

public class LRUCache {
    private static class Node {
        int key, value;
        Node prev, next;
        Node(int k, int v) { this.key = k; this.value = v; }
    }

    private final int capacity;
    private final Map<Integer, Node> map = new HashMap<>();
    private final Node head = new Node(0, 0); // Most recently used
    private final Node tail = new Node(0, 0); // Least recently used

    public LRUCache(int capacity) {
        this.capacity = capacity;
        head.next = tail;
        tail.prev = head;
    }

    public int get(int key) {
        if (!map.containsKey(key)) return -1;
        Node node = map.get(key);
        moveToHead(node); // Mark as recently used
        return node.value;
    }

    public void put(int key, int value) {
        if (map.containsKey(key)) {
            Node node = map.get(key);
            node.value = value;
            moveToHead(node);
        } else {
            if (map.size() == capacity) {
                // Evict least recently used (node right before tail)
                Node lru = tail.prev;
                removeNode(lru);
                map.remove(lru.key);
            }
            Node newNode = new Node(key, value);
            map.put(key, newNode);
            addToHead(newNode);
        }
    }

    private void addToHead(Node node) {
        node.next = head.next;
        node.prev = head;
        head.next.prev = node;
        head.next = node;
    }

    private void removeNode(Node node) {
        node.prev.next = node.next;
        node.next.prev = node.prev;
    }

    private void moveToHead(Node node) {
        removeNode(node);
        addToHead(node);
    }
}
